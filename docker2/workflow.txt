→ create a Dockerfile with:
FROM nginx:latest
COPY index.html /usr/share/nginx/html/index.html
EXPOSE 80
CMD ["nginx", "-g", "daemon off;"]

→ create Docker compose file with:
version: "3.8"
services:
  web:
    build: .
    container_name: nginx-web-app
    ports:
      - "8080:80"
    restart: always


→ docker compose build
→ docker images
→ docker compose up -d
→ docker ps
→ curl localhost:8080 you can see the result where whole html document will return
→ http://<aws-pub-ip>:8080
